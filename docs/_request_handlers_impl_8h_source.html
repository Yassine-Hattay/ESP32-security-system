182 bool handle(WebServer &server, HTTPMethod requestMethod, const String &requestUri) override {
183 if (!canHandle(server, requestMethod, requestUri)) {
184 return false;
185 }
186
187 log_v("StaticRequestHandler::handle: request=%s _uri=%s\r\n", requestUri.c_str(), _uri.c_str());
188
189 String path(_path);
190
191 if (!_isFile) {
192 // Base URI doesn't point to a file.
193 // If a directory is requested, look for index file.
194 if (requestUri.endsWith("/")) {
195 return handle(server, requestMethod, String(requestUri + "index.htm"));
196 }
197
198 // Append whatever follows this URI in request to get the file path.
199 path += requestUri.substring(_baseUriLength);
200 }
201 log_v("StaticRequestHandler::handle: path=%s, isFile=%d\r\n", path.c_str(), _isFile);
202
203 String contentType = getContentType(path);
204
205 // look for gz file, only if the original specified path is not a gz. So part only works to send gzip via content encoding when a non compressed is asked for
206 // if you point the the path to gzip you will serve the gzip as content type "application/x-gzip", not text or javascript etc...
207 if (!path.endsWith(FPSTR(mimeTable[gz].endsWith)) && !_fs.exists(path)) {
208 String pathWithGz = path + FPSTR(mimeTable[gz].endsWith);
209 if (_fs.exists(pathWithGz)) {
210 path += FPSTR(mimeTable[gz].endsWith);
211 }
212 }
213
214 File f = _fs.open(path, "r");
215 if (!f || !f.available()) {
216 return false;
217 }
218
219 String eTagCode;
220
221 if (server._eTagEnabled) {
222 if (server._eTagFunction) {
223 eTagCode = (server._eTagFunction)(_fs, path);
224 } else {
225 eTagCode = calcETag(_fs, path);
226 }
227
228 if (server.header("If-None-Match") == eTagCode) {
229 server.send(304);
230 return true;
231 }
232 }
233
234 if (_cache_header.length() != 0) {
235 server.sendHeader("Cache-Control", _cache_header);
236 }
237
238 if ((server._eTagEnabled) && (eTagCode.length() > 0)) {
239 server.sendHeader("ETag", eTagCode);
240 }
241
242 server.streamFile(f, contentType);
243 return true;
244 }

246 static String getContentType(const String &path) {
247 char buff[sizeof(mimeTable[0].mimeType)];
248 // Check all entries but last one for match, return if found
249 for (size_t i = 0; i < sizeof(mimeTable) / sizeof(mimeTable[0]) - 1; i++) {
250 strcpy_P(buff, mimeTable[i].endsWith);
251 if (path.endsWith(buff)) {
252 strcpy_P(buff, mimeTable[i].mimeType);
253 return String(buff);
254 }
255 }
256 // Fall-through and just return default type
257 strcpy_P(buff, mimeTable[sizeof(mimeTable) / sizeof(mimeTable[0]) - 1].mimeType);
258 return String(buff);
259 }

261 // calculate an ETag for a file in filesystem based on md5 checksum
262 // that can be used in the http headers - include quotes.
263 static String calcETag(FS &fs, const String &path) {
264 String result;
265
266 // calculate eTag using md5 checksum
267 uint8_t md5_buf[16];
268 File f = fs.open(path, "r");
269 MD5Builder calcMD5;
270 calcMD5.begin();
271 calcMD5.addStream(f, f.size());
272 calcMD5.calculate();
273 calcMD5.getBytes(md5_buf);
274 f.close();
275 // create a minimal-length eTag using base64 byte[]->text encoding.
276 result = "\"" + base64::encode(md5_buf, 16) + "\"";
277 return (result);
278 } // calcETag
