325 // BearSSL doesn't define a true insecure decoder, so we make one ourselves
326 // from the simple parser. It generates the issuer and subject hashes and
327 // the SHA1 fingerprint, only one (or none!) of which will be used to
328 // "verify" the certificate.
329
330 // Private x509 decoder state
331 struct br_x509_insecure_context
332 {
333 const br_x509_class *vtable;
334 bool done_cert;
335 const uint8_t *match_fingerprint;
336 br_sha1_context sha1_cert;
337 bool allow_self_signed;
338 br_sha256_context sha256_subject;
339 br_sha256_context sha256_issuer;
340 br_x509_decoder_context ctx;
341 };
342
343 // Callback for the x509_minimal subject DN
344 static void insecure_subject_dn_append(void *ctx, const void *buf, size_t len)
345 {
346 br_x509_insecure_context *xc = reinterpret_cast<br_x509_insecure_context *>(ctx);
347 br_sha256_update(&xc->sha256_subject, buf, len);
348 }
349
350 // Callback for the x509_minimal issuer DN
351 static void insecure_issuer_dn_append(void *ctx, const void *buf, size_t len)
352 {
353 br_x509_insecure_context *xc = reinterpret_cast<br_x509_insecure_context *>(ctx);
354 br_sha256_update(&xc->sha256_issuer, buf, len);
355 }
356
357 // Callback for each certificate present in the chain (but only operates
358 // on the first one by design).
359 static void insecure_start_cert(const br_x509_class **ctx, uint32_t length)
360 {
361 (void)ctx;
362 (void)length;
363 }
364
365 // Callback for each byte stream in the chain. Only process first cert.
366 static void insecure_append(const br_x509_class **ctx, const unsigned char *buf, size_t len)
367 {
368 br_x509_insecure_context *xc = reinterpret_cast<br_x509_insecure_context *>(ctx);
369 // Don't process anything but the first certificate in the chain
370 if (!xc->done_cert)
371 {
372 br_sha1_update(&xc->sha1_cert, buf, len);
373 br_x509_decoder_push(&xc->ctx, reinterpret_cast<const void *>(buf), len);
374 }
375 }
376 // Callback on the first byte of any certificate
377 static void insecure_start_chain(const br_x509_class **ctx, const char *server_name)
378 {
379 br_x509_insecure_context *xc = reinterpret_cast<br_x509_insecure_context *>(ctx);
380#if defined(USE_EMBED_SSL_ENGINE)
381 br_x509_decoder_init(&xc->ctx, insecure_subject_dn_append, xc, insecure_issuer_dn_append, xc);
382#elif defined(ESP32) || defined(USE_LIB_SSL_ENGINE)
383 br_x509_decoder_init(&xc->ctx, insecure_subject_dn_append, xc);
384#endif
385 xc->done_cert = false;
386 br_sha1_init(&xc->sha1_cert);
387 br_sha256_init(&xc->sha256_subject);
388 br_sha256_init(&xc->sha256_issuer);
389 (void)server_name;
390 }
391
392 // Callback on individual cert end.
393 static void insecure_end_cert(const br_x509_class **ctx)
394 {
395 br_x509_insecure_context *xc = reinterpret_cast<br_x509_insecure_context *>(ctx);
396 xc->done_cert = true;
397 }
398
399 // Callback when complete chain has been parsed.
400 // Return 0 on validation success, !0 on validation error
401 static unsigned insecure_end_chain(const br_x509_class **ctx)
402 {
403 const br_x509_insecure_context *xc = reinterpret_cast<const br_x509_insecure_context *>(ctx);
404 if (!xc->done_cert)
405 {
406 // BSSL_BSSL_SSL_Client_DEBUG_PRINTF("insecure_end_chain: No cert seen\n");
407 return 1; // error
408 }
409
410 // Handle SHA1 fingerprint matching
411 char res[20];
412 br_sha1_out(&xc->sha1_cert, res);
413 if (xc->match_fingerprint && memcmp(res, xc->match_fingerprint, sizeof(res)))
414 {
415
416 return BR_ERR_X509_NOT_TRUSTED;
417 }
418
419 // Handle self-signer certificate acceptance
420 char res_issuer[32];
421 char res_subject[32];
422 br_sha256_out(&xc->sha256_issuer, res_issuer);
423 br_sha256_out(&xc->sha256_subject, res_subject);
424 if (xc->allow_self_signed && memcmp(res_subject, res_issuer, sizeof(res_issuer)))
425 {
426 // BSSL_BSSL_SSL_Client_DEBUG_PRINTF("insecure_end_chain: Didn't get self-signed cert\n");
427 return BR_ERR_X509_NOT_TRUSTED;
428 }
429
430 // Default (no validation at all) or no errors in prior checks = success.
431 return 0;
432 }
433
434 // Return the public key from the validator (set by x509_minimal)
435 static const br_x509_pkey *insecure_get_pkey(const br_x509_class *const *ctx, unsigned *usages)
436 {
437 const br_x509_insecure_context *xc = reinterpret_cast<const br_x509_insecure_context *>(ctx);
438 if (usages != nullptr)
439 {
440 *usages = BR_KEYTYPE_KEYX | BR_KEYTYPE_SIGN; // I said we were insecure!
441 }
442 return &xc->ctx.pkey;
443 }
444 }
