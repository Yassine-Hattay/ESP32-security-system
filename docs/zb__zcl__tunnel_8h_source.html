149#define ZB_ZCL_TUNNEL_SEND_TRANSFER_REQ( \
150 _buffer, _addr, _dst_addr_mode, _dst_ep, _ep, _prfl_id, _def_resp, _cb, \
151 _manufacturer_id, _flag, _byte_num, _data_size, _image_data) \
152{ \
153 zb_uint8_t* __ptr = ZB_ZCL_START_PACKET((_buffer)); \
154 ZB_ZCL_CONSTRUCT_SPECIFIC_COMMAND_REQ_FRAME_CONTROL_A(__ptr, \
155 ZB_ZCL_FRAME_DIRECTION_TO_SRV, ZB_ZCL_MANUFACTURER_SPECIFIC, (_def_resp));\
156 ZB_ZCL_CONSTRUCT_COMMAND_HEADER_EXT(__ptr, ZB_ZCL_GET_SEQ_NUM(), \
157 ZB_ZCL_MANUFACTURER_SPECIFIC, (_manufacturer_id), \
158 ZB_ZCL_CMD_TUNNEL_TRANSFER_DATA_REQ); \
159 ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_flag)); \
160 ZB_ZCL_PACKET_PUT_DATA16_VAL(__ptr, (_byte_num)); \
161 ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_data_size)); \
162 if((_data_size)>0) \
163 { \
164 ZB_ZCL_PACKET_PUT_DATA_N(__ptr, (_image_data), (_data_size)); \
165 } \
166 ZB_ZCL_FINISH_PACKET((_buffer), __ptr) \
167 ZB_ZCL_SEND_COMMAND_SHORT( \
168 (_buffer), (_addr), (_dst_addr_mode), (_dst_ep), (_ep), (_prfl_id), \
169 ZB_ZCL_CLUSTER_ID_TUNNEL, (_cb)); \
170}
171
181#define ZB_ZCL_TUNNEL_GET_TRANSFER_REQ(_data_ptr, _buffer, _status) \
182{ \
183 zb_zcl_tunnel_transfer_data_req_t *src_ptr = \
184 (zb_zcl_tunnel_transfer_data_req_t*)zb_buf_begin((_buffer)); \
185 \
186 if (zb_buf_len((_buffer)) != sizeof(zb_zcl_tunnel_transfer_data_req_t) \
187 + src_ptr->data_size * sizeof(zb_uint8_t) ) \
188 { \
189 (_status) = ZB_ZCL_PARSE_STATUS_FAILURE; \
190 } \
191 else \
192 { \
193 (_status) = ZB_ZCL_PARSE_STATUS_SUCCESS; \
194 (_data_ptr)->req_header.tx_flag = src_ptr->tx_flag; \
195 ZB_HTOLE16(&((_data_ptr)->req_header.byte_num), &(src_ptr->byte_num)); \
196 (_data_ptr)->req_header.data_size = src_ptr->data_size; \
197 (_data_ptr)->tun_data = (zb_uint8_t*)(src_ptr+1); \
198 } \
199}

232 ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_tun_status)); \
233 ZB_ZCL_FINISH_PACKET((_buffer), __ptr) \

236 ZB_ZCL_CLUSTER_ID_TUNNEL, (_cb)); \
