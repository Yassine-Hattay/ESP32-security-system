174#define ZB_ZCL_BINARY_INPUT_SET_NORMAL_MODE(ep) \
175{ \
176 zb_uint8_t val; \
177 \
178 val = ZB_FALSE; \
179 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_BINARY_INPUT, \
180 ZB_ZCL_CLUSTER_SERVER_ROLE, \
181 ZB_ZCL_ATTR_BINARY_INPUT_OUT_OF_SERVICE_ID, \
182 &val, ZB_FALSE); \
183 \
184 val = ZB_ZCL_BINARY_INPUT_STATUS_FLAG_NORMAL; \
185 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_BINARY_INPUT, \
186 ZB_ZCL_CLUSTER_SERVER_ROLE, \
187 ZB_ZCL_ATTR_BINARY_INPUT_STATUS_FLAG_ID, \
188 &val, ZB_FALSE); \
189}

194#define ZB_ZCL_BINARY_INPUT_SET_OUT_OF_SERVICE(ep) \
195{ \
196 zb_zcl_attr_t *attr_desc; \
197 zb_uint8_t val; \
198 \
199 val = ZB_TRUE; \
200 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_BINARY_INPUT, \
201 ZB_ZCL_CLUSTER_SERVER_ROLE, \
202 ZB_ZCL_ATTR_BINARY_INPUT_OUT_OF_SERVICE_ID, \
203 &val, ZB_FALSE); \
204 \
205 attr_desc = zb_zcl_get_attr_desc_a(ep, ZB_ZCL_CLUSTER_ID_BINARY_INPUT, \
206 ZB_ZCL_CLUSTER_SERVER_ROLE, \
207 ZB_ZCL_ATTR_BINARY_INPUT_STATUS_FLAG_ID); \
208 if (attr_desc) \
209 { \
210 val = *(zb_uint8_t*)attr_desc->data_p | ZB_ZCL_BINARY_INPUT_STATUS_FLAG_OUT_OF_SERVICE; \
211 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_BINARY_INPUT, \
212 ZB_ZCL_CLUSTER_SERVER_ROLE, \
213 ZB_ZCL_ATTR_BINARY_INPUT_STATUS_FLAG_ID, \
214 &val, ZB_FALSE); \
215 } \
216}
