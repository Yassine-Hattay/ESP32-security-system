131#define ZB_ADDRESS_DECOMPRESS(address, compressed_address) \
132do \
133{ \
134 if (ZB_ADDRESS_COMPRESSED_IS_UNKNOWN(compressed_address)) \
135 { \
136 ZB_64BIT_ADDR_UNKNOWN(address); \
137 } \
138 else \
139 { \
140 ZB_MEMCPY(&((address)[5]), \
141 &(ZG->addr.dev_manufacturer[(compressed_address).dev_manufacturer].device_manufacturer[0]), \
142 (sizeof((address)[0]) * 3U)); \
143 ZB_MEMCPY(&((address)[0]), &((compressed_address).device_id[0]), (sizeof((address)[0]) * 5U)); \
144 \
145 } \
146} \
147while (0)
