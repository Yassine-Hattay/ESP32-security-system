313/*
314 * start_chain Called when a new chain is started. If 'server_name'
315 * is not NULL and non-empty, then it is a name that
316 * should be looked for in the EE certificate (in the
317 * SAN extension as dNSName, or in the subjectDN's CN
318 * if there is no SAN extension).
319 * The caller ensures that the provided 'server_name'
320 * pointer remains valid throughout validation.
321 *
322 * start_cert Begins a new certificate in the chain. The provided
323 * length is in bytes; this is the total certificate length.
324 *
325 * append Get some additional bytes for the current certificate.
326 *
327 * end_cert Ends the current certificate.
328 *
329 * end_chain Called at the end of the chain. Returned value is
330 * 0 on success, or a non-zero error code.
331 *
332 * get_pkey Returns the EE certificate public key.
333 *
334 * For a complete chain, start_chain() and end_chain() are always
335 * called. For each certificate, start_cert(), some append() calls, then
336 * end_cert() are called, in that order. There may be no append() call
337 * at all if the certificate is empty (which is not valid but may happen
338 * if the peer sends exactly that).
339 *
340 * get_pkey() shall return a pointer to a structure that is valid as
341 * long as a new chain is not started. This may be a sub-structure
342 * within the context for the engine. This function MAY return a valid
343 * pointer to a public key even in some cases of validation failure,
344 * depending on the validation engine.
345 */

530/*
531 * The minimal X.509 engine has some state buffers which must be large
532 * enough to simultaneously accommodate:
533 * -- the public key extracted from the current certificate;
534 * -- the signature on the current certificate or on the previous
535 * certificate;
536 * -- the public key extracted from the EE certificate.
537 *
538 * We store public key elements in their raw unsigned big-endian
539 * encoding. We want to support up to RSA-4096 with a short (up to 64
540 * bits) public exponent, thus a buffer for a public key must have
541 * length at least 520 bytes. Similarly, a RSA-4096 signature has length
542 * 512 bytes.
543 *
544 * Though RSA public exponents can formally be as large as the modulus
545 * (mathematically, even larger exponents would work, but PKCS#1 forbids
546 * them), exponents that do not fit on 32 bits are extremely rare,
547 * notably because some widespread implementations (e.g. Microsoft's
548 * CryptoAPI) don't support them. Moreover, large public exponent do not
549 * seem to imply any tangible security benefit, and they increase the
550 * cost of public key operations. The X.509 "minimal" engine will tolerate
551 * public exponents of arbitrary size as long as the modulus and the
552 * exponent can fit together in the dedicated buffer.
553 *
554 * EC public keys are shorter than RSA public keys; even with curve
555 * NIST P-521 (the largest curve we care to support), a public key is
556 * encoded over 133 bytes only.
557 */
558#define BR_X509_BUFSIZE_KEY 520
559#define BR_X509_BUFSIZE_SIG 512
560#endif
