2052/*
2053 * There are four I/O operations, each identified by a symbolic name:
2054 *
2055 * sendapp inject application data in the engine
2056 * recvapp retrieving application data from the engine
2057 * sendrec sending records on the transport medium
2058 * recvrec receiving records from the transport medium
2059 *
2060 * Terminology works thus: in a layered model where the SSL engine sits
2061 * between the application and the network, "send" designates operations
2062 * where bytes flow from application to network, and "recv" for the
2063 * reverse operation. Application data (the plaintext that is to be
2064 * conveyed through SSL) is "app", while encrypted records are "rec".
2065 * Note that from the SSL engine point of view, "sendapp" and "recvrec"
2066 * designate bytes that enter the engine ("inject" operation), while
2067 * "recvapp" and "sendrec" designate bytes that exit the engine
2068 * ("extract" operation).
2069 *
2070 * For the operation 'xxx', two functions are defined:
2071 *
2072 * br_ssl_engine_xxx_buf
2073 * Returns a pointer and length to the buffer to use for that
2074 * operation. '*len' is set to the number of bytes that may be read
2075 * from the buffer (extract operation) or written to the buffer
2076 * (inject operation). If no byte may be exchanged for that operation
2077 * at that point, then '*len' is set to zero, and NULL is returned.
2078 * The engine state is unmodified by this call.
2079 *
2080 * br_ssl_engine_xxx_ack
2081 * Informs the engine that 'len' bytes have been read from the buffer
2082 * (extract operation) or written to the buffer (inject operation).
2083 * The 'len' value MUST NOT be zero. The 'len' value MUST NOT exceed
2084 * that which was obtained from a preceding br_ssl_engine_xxx_buf()
2085 * call.
2086 */

3952/*
3953 * Context for the simplified I/O context. The transport medium is accessed
3954 * through the low_read() and low_write() callback functions, each with
3955 * its own opaque context pointer.
3956 *
3957 * low_read() read some bytes, at most 'len' bytes, into data[]. The
3958 * returned value is the number of read bytes, or -1 on error.
3959 * The 'len' parameter is guaranteed never to exceed 20000,
3960 * so the length always fits in an 'int' on all platforms.
3961 *
3962 * low_write() write up to 'len' bytes, to be read from data[]. The
3963 * returned value is the number of written bytes, or -1 on
3964 * error. The 'len' parameter is guaranteed never to exceed
3965 * 20000, so the length always fits in an 'int' on all
3966 * parameters.
3967 *
3968 * A socket closure (if the transport medium is a socket) should be reported
3969 * as an error (-1). The callbacks shall endeavour to block until at least
3970 * one byte can be read or written; a callback returning 0 at times is
3971 * acceptable, but this normally leads to the callback being immediately
3972 * called again, so the callback should at least always try to block for
3973 * some time if no I/O can take place.
3974 *
3975 * The SSL engine naturally applies some buffering, so the callbacks need
3976 * not apply buffers of their own.
3977 */

3985#ifndef BR_DOXYGEN_IGNORE
