319/* EE: wrong place for that constant in zb_aps.h (this is not public APS API!) and wrong
320 * calculation!
321 * BTW that constant used at 1 place only.
322 * ZB_ZDO_MAX_PAYLOAD_SIZE in zb_zdo.h is hard-coded!
323 * And, zb_zdo.h is also bad place for it.
324 *
325 * Let's count supposing no APS security.
326 * MAC header - 10 with fcf (suppose Pan ID compression)
327 * NWK header - 24 unicast, but 16 broadcast
328 * NWK ZB security header + MIC - cf 1 + fcnt 4 + ext src 8 + key seq 1 mic 4 = 18
329 * APS hdr 8b
330 * Totally for broadcast - 127 - (11+16+18+8) = 74
331 * For unicast 74-8 = 66
332
333Fixed that size for the certification test TP/R21/BV-24.
334We actually fit 5 parent annce records into 127 bytes only for broadcast tx,
335with Pan ID compression.
336
337At the worst case our NWK can skip long address at tx: 8 bytes of reserve.
338
339 */

352/* ZB packet length must not exceed 127 bytes
353 *
354 * Old calculation by Maxim:
355 *
356 * MAC header (total: 11 bytes)
357 * - FCF: 2 bytes
358 * - Sequence number: 1 bytes
359 * - PAN: 2 bytes
360 * - Destination address: 2 bytes
361 * - Source address: 2 bytes
362 *
363 * NWK header (total: 24 bytes)
364 * - FCF: 2 bytes
365 * - Destination short address: 2 bytes
366 * - Source short address: 2 bytes
367 * - Radius: 1 byte
368 * - Sequence number: 1 byte
369 * - Destination long address: 8 bytes
370 * - Source long address: 8 bytes
371 *
372 * ZB security header (total: 31 bytes)
373 *
374 * APS header (total: 8 bytes)
375 * - FCF: 1 byte
376 * - Destination endpoint: 1 byte
377 * - Cluster ID: 2 bytes
378 * - Profile ID: 2 bytes
379 * - Source endpoint: 1 byte
380 * - Counter: 1 byte
381 *
382 * APS payload (total: variable length)
383 *
384 * MAC footer (total: 2 bytes)
385 * - FCS: 2 bytes
386 *
387 * Total length w/o APS payload: 127-(11+24+31+8+2)= 51 bytes,
388 * consequently APS payload max size is 51 bytes.
389 */
390/*
391 * MAC (short source, short destination): MAX_MAC_OVERHEAD_SHORT_ADDRS bytes
392 * NWK (broadcast, no destination IEEE): 16 bytes header + 18 bytes security
393 * Total length: 127-(11+16+18) = 82 bytes */
394 /* nwk hdr include source IEEE address */

517/* APS base header: 8b
518 * 1b - FCF
519 * 1b - dst ep
520 * 2b - cluster id
521 * 2b - profile id
522 * 1b - src ep
523 * 1b - counter
524 *
525 * ext FC: 2b
526 * 1b - fragmentation
527 * 1b - block number
528 * 1b - ACK bitfield - it doesn't matter for data packets
529 *
530 * security: 17b
531 * 1b - control field
532 * 4b - frame counter
533 * 4b - message integrity code
534 * 8b - ieee addr
535 *
536 * total: 8b + 2b + 17b = 27b
537 */
