45// Each RMT Symbols has 4 bytes
46// Total number of bytes per RMT_MEM_BLOCK is RMT_SYMBOLS_PER_CHANNEL_BLOCK * 4 bytes
47typedef union {
48 struct {
49 uint32_t duration0 : 15;
50 uint32_t level0 : 1;
51 uint32_t duration1 : 15;
52 uint32_t level1 : 1;
53 };
54 uint32_t val;
55} rmt_data_t;
56
57// Reading and Writing shall use as rmt_symbols_size this unit
58// ESP32 has 8 MEM BLOCKS in total shared with Reading and/or Writing
59// ESP32-S2 has 4 MEM BLOCKS in total shared with Reading and/or Writing
60// ESP32-S3 has 4 MEM BLOCKS for Reading and another 4 MEM BLOCKS for Writing
61// ESP32-C3 has 2 MEM BLOCKS for Reading and another 2 MEM BLOCKS for Writing
62#define RMT_SYMBOLS_PER_CHANNEL_BLOCK SOC_RMT_MEM_WORDS_PER_CHANNEL

64// Used to tell rmtRead() to wait for ever until reading data from the RMT channel

67// Helper macro to calculate the number of RTM symbols in a array or type
68#define RMT_SYMBOLS_OF(x) (sizeof(x) / sizeof(rmt_data_t))
