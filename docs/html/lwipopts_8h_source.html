127/* ---------- Memory options ---------- */
128/* MEM_ALIGNMENT: should be set to the alignment of the CPU for which
129 lwIP is compiled. 4 byte alignment -> define MEM_ALIGNMENT to 4, 2
130 byte alignment -> define MEM_ALIGNMENT to 2. */
131/* MSVC port: intel processors don't need 4-byte alignment,
132 but are faster that way! */
133#define MEM_ALIGNMENT 4U
134
135/* MEM_SIZE: the size of the heap memory. If the application will send
136a lot of data that needs to be copied, this should be set high. */
137#define MEM_SIZE 10240
138
139/* MEMP_NUM_PBUF: the number of memp struct pbufs. If the application
140 sends a lot of data out of ROM (or other static memory), this
141 should be set high. */
142#define MEMP_NUM_PBUF 16
143/* MEMP_NUM_RAW_PCB: the number of UDP protocol control blocks. One
144 per active RAW "connection". */
145#define MEMP_NUM_RAW_PCB 3
146/* MEMP_NUM_UDP_PCB: the number of UDP protocol control blocks. One
147 per active UDP "connection". */
148#define MEMP_NUM_UDP_PCB 4
149/* MEMP_NUM_TCP_PCB: the number of simulatenously active TCP
150 connections. */
151#define MEMP_NUM_TCP_PCB 5
152/* MEMP_NUM_TCP_PCB_LISTEN: the number of listening TCP
153 connections. */
154#define MEMP_NUM_TCP_PCB_LISTEN 8
155/* MEMP_NUM_TCP_SEG: the number of simultaneously queued TCP
156 segments. */
157#define MEMP_NUM_TCP_SEG 16
158/* MEMP_NUM_SYS_TIMEOUT: the number of simulateously active
159 timeouts. */
160#define MEMP_NUM_SYS_TIMEOUT 17
161
162/* The following four are used only with the sequential API and can be
163 set to 0 if the application only will use the raw API. */
164/* MEMP_NUM_NETBUF: the number of struct netbufs. */
165#define MEMP_NUM_NETBUF 2
166/* MEMP_NUM_NETCONN: the number of struct netconns. */
167#define MEMP_NUM_NETCONN 10
168/* MEMP_NUM_TCPIP_MSG_*: the number of struct tcpip_msg, which is used
169 for sequential API communication and incoming packets. Used in
170 src/api/tcpip.c. */
171#define MEMP_NUM_TCPIP_MSG_API 16
172#define MEMP_NUM_TCPIP_MSG_INPKT 16
173

211/* TCP sender buffer space (pbufs). This must be at least = 2 *
212 TCP_SND_BUF/TCP_MSS for things to work. */
213#define TCP_SND_QUEUELEN (4 * TCP_SND_BUF/TCP_MSS)

216 to TCP_SND_BUF. It is the amount of space which must be
217 available in the tcp snd_buf for select to return writable */
218#define TCP_SNDLOWAT (TCP_SND_BUF/2)
