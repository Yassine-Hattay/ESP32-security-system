526typedef ZB_PACKED_PRE struct zb_mac_diagnostic_info_s
527{
528 zb_uint32_t mac_rx_bcast; /* A counter that is incremented each time
529 * the MAC layer receives a broadcast. */
530 zb_uint32_t mac_tx_bcast; /* A counter that is incremented each time
531 * the MAC layer transmits a broadcast. */
532 zb_uint32_t mac_rx_ucast; /* A counter that is incremented each time the
533 * MAC layer receives a unicast. */
534
535 /* These 3 counters are required not to break
536 * ZDO channel management logic that
537 * uses normalized counters values.
538 */
539 zb_uint32_t mac_tx_ucast_total_zcl; /* The same as mac_tx_ucast_total, but non-normalized */
540 zb_uint16_t mac_tx_ucast_failures_zcl; /* The same as mac_tx_ucast_failures, but non-normalized */
541 zb_uint16_t mac_tx_ucast_retries_zcl; /* The same as mac_tx_ucast_retries, but non-normalized*/
542
543 zb_uint16_t mac_tx_ucast_total; /* Total number of Mac Tx Transactions to
544 * attempt to send a message (but not
545 * counting retries) */
546 zb_uint16_t mac_tx_ucast_failures; /* Total number of failed Tx
547 * Transactions. So if the Mac send a
548 * single packet, it will be retried 4
549 * times without ack, that counts as 1 failure */
550 zb_uint16_t mac_tx_ucast_retries; /* Total number of Mac Retries regardless of
551 * whether the transaction resulted in
552 * success or failure. */
553
554 zb_uint16_t phy_to_mac_que_lim_reached; /* A counter that is incremented each time when MAC RX queue if full. */
555
556 zb_uint16_t mac_validate_drop_cnt; /* How many times the packet was dropped at the packet
557 * validation stage for length or bad formatting. */
558
559 zb_uint16_t phy_cca_fail_count; /* Number of the PHY layer was unable
560 * to transmit due to a failed CCA */
561
562 zb_uint8_t period_of_time; /* Time period over which MACTx results are measured */
563 zb_uint8_t last_msg_lqi; /* LQI value of the last received packet */
564 zb_int8_t last_msg_rssi; /* RSSI value of the last received packet */
565 zb_uint32_t cca_retries; /* Total number of CCA retries */
566 zb_uint32_t pta_lo_pri_req; /* Total number of low priority PTA request */
567 zb_uint32_t pta_hi_pri_req; /* Total number of high priority PTA request */
568 zb_uint32_t pta_lo_pri_denied; /* Total number of low priority PTA request denied by master */
569 zb_uint32_t pta_hi_pri_denied; /* Total number of high priority PTA request denied by master */
570 zb_uint32_t pta_denied_rate; /* PTA deny rate*/
571} ZB_PACKED_STRUCT
