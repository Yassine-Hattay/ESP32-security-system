108#define ZB_ZCL_IR_BLASTER_SEND_TRANSMIT_IR_DATA( \
109 _buffer, _addr, _dst_addr_mode, _dst_ep, _ep, _prfl_id, _def_resp, _cb, \
110 _manufacturer_id, _ir_code_id) \
111{ \
112 zb_uint8_t* __ptr = ZB_ZCL_START_PACKET((_buffer)); \
113 ZB_ZCL_CONSTRUCT_SPECIFIC_COMMAND_REQ_FRAME_CONTROL_A(__ptr, \
114 ZB_ZCL_FRAME_DIRECTION_TO_SRV, ZB_ZCL_MANUFACTURER_SPECIFIC, (_def_resp)); \
115 ZB_ZCL_CONSTRUCT_COMMAND_HEADER_EXT(__ptr, ZB_ZCL_GET_SEQ_NUM(), \
116 ZB_ZCL_MANUFACTURER_SPECIFIC, (_manufacturer_id), \
117 ZB_ZCL_CMD_IR_BLASTER_TRANSMIT_IR_DATA); \
118 ZB_ZCL_PACKET_PUT_DATA_N(__ptr, (_ir_code_id), \
119 (sizeof(zb_zcl_ir_blaster_transmit_ir_data_t))); \
120 ZB_ZCL_FINISH_PACKET((_buffer), __ptr) \
121 ZB_ZCL_SEND_COMMAND_SHORT( \
122 (_buffer), (_addr), (_dst_addr_mode), (_dst_ep), (_ep), (_prfl_id), \
123 ZB_ZCL_CLUSTER_ID_IR_BLASTER, (_cb)); \
124}

275#define ZB_ZCL_IR_BLASTER_SEND_GET_IR_SIGNATURE_RESP( \
276 _buffer, _addr, _dst_addr_mode, _dst_ep, _ep, _prfl_id, _seq, _cb, \
277 _manufacturer_id, _status, _ir_signature) \
278{ \
279 zb_uint8_t* __ptr = ZB_ZCL_START_PACKET((_buffer)); \
280 ZB_ZCL_CONSTRUCT_SPECIFIC_COMMAND_RESP_FRAME_CONTROL_A(__ptr, \

284 ZB_ZCL_PACKET_PUT_DATA8(__ptr, _status); \
285 if (_status == ZB_ZCL_IR_BLASTER_TRANSMISSION_STATUS_COMPLETED) \
286 { \
287 ZB_ZCL_PACKET_PUT_DATA_N(__ptr, (_ir_signature), ZB_ZCL_IR_BLASTER_IR_SIGNATURE_SIZE); \
288 } \
289 ZB_ZCL_FINISH_PACKET((_buffer), __ptr) \
290 ZB_ZCL_SEND_COMMAND_SHORT( \
