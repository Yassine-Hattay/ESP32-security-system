848/* Update a parameter
849 *
850 * This will just update the value of a parameter with esp rainmaker core, without actually reporting
851 * it. This can be used when multiple parameters need to be reported together.
852 * Eg. If x parameters are to be reported, this API can be used for the first x -1 parameters
853 * and the last one can be updated using esp_rmaker_param_update_and_report().
854 * This will report all parameters which were updated prior to this call.
855 *
856 * @note This does not report to time series even if PROP_FLAG_TIME_SERIES is set.
857 *
858 * Sample:
859 *
860 * esp_rmaker_param_update(param1, esp_rmaker_float(10.2));
861 * esp_rmaker_param_update(param2, esp_rmaker_int(55));
862 * esp_rmaker_param_update(param3, esp_rmaker_int(95));
863 * esp_rmaker_param_update_and_report(param1, esp_rmaker_bool(true));
864 *
865 * @param[in] param Parameter handle.
866 * @param[in] val New value of the parameter.
867 *
868 * @return ESP_OK if the parameter was updated successfully.
869 * @return error in case of failure.
870 */
871esp_err_t esp_rmaker_param_update(const esp_rmaker_param_t *param, esp_rmaker_param_val_t val);
