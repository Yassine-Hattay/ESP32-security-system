194 /*
195 Template to write from objects that expose
196 available() and read(uint8_t*, size_t) methods
197 faster than the writeStream method
198 writes only what is available
199 */
200 template<typename T> size_t write(T &data) {
201 size_t written = 0;
202 if (hasError() || !isRunning()) {
203 return 0;
204 }
205
206 size_t available = data.available();
207 while (available) {
208 if (_bufferLen + available > remaining()) {
209 available = remaining() - _bufferLen;
210 }
211 if (_bufferLen + available > 4096) {
212 size_t toBuff = 4096 - _bufferLen;
213 data.read(_buffer + _bufferLen, toBuff);
214 _bufferLen += toBuff;
215 if (!_writeBuffer()) {
216 return written;
217 }
218 written += toBuff;
219 } else {
220 data.read(_buffer + _bufferLen, available);
221 _bufferLen += available;
222 written += available;
223 if (_bufferLen == remaining()) {
224 if (!_writeBuffer()) {
225 return written;
226 }
227 }
228 }
229 if (remaining() == 0) {
230 return written;
231 }
232 available = data.available();
233 }
234 return written;
235 }
