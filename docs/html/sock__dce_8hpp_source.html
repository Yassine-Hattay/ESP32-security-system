124 bool wait_to_idle(uint32_t ms)
125 {
126 if (!signal.wait(IDLE, ms)) {
127 ESP_LOGE("dce", "Failed to get idle");
128 return false;
129 }
130 if (state != status::IDLE) {
131 ESP_LOGE("dce", "Unexpected state %d", static_cast<int>(state));
132 return false;
133 }
134 return true;
135 }

137 int sync_recv(char *buffer, int len, int timeout_ms)
138 {
139 if (!wait_to_idle(timeout_ms)) {
140 return 0;
141 }
142 at.clear_offsets();
143 state = status::RECEIVING;
144 uint64_t data;
145 read(data_ready_fd, &data, sizeof(data));
146 int max_len = std::min(len, (int)at.get_buf_len());
147 at.start_receiving(max_len);
148 if (!signal.wait(IDLE, 500 + timeout_ms)) {
149 return 0;
150 }
151 int ret = at.data_available();
152 if (ret > 0) {
153 memcpy(buffer, at.get_buf(), ret);
154 }
155 set_idle();
156 return ret;
157 }
158
159 int sync_send(const char *buffer, size_t len, int timeout_ms)
160 {
161 int len_to_send = std::min(len, at.get_buf_len());
162 if (!wait_to_idle(timeout_ms)) {
163 return -1;
164 }
165 state = status::SENDING;
166 memcpy(at.get_buf(), buffer, len_to_send);
167 ESP_LOG_BUFFER_HEXDUMP("dce", at.get_buf(), len, ESP_LOG_VERBOSE);
168 at.start_sending(len_to_send);
169 if (!signal.wait(IDLE, timeout_ms + 1000)) {
170 if (state == status::PENDING) {
171 state = status::IDLE;
172 } else {
173 return -1;
174 }
175 }
176 set_idle();
177 return len_to_send;
178 }
179
180 int wait_to_read(uint32_t ms)
181 {
182 if (at.has_data() > 0) {
183 ESP_LOGD("dce", "Data buffered in modem (len=%d)", at.has_data());
184 return 1;
185 }
186 struct timeval tv = {
187 .tv_sec = static_cast<time_t>(ms / 1000),
188 .tv_usec = 0,
189 };
190 fd_set fdset;
191 FD_ZERO(&fdset);
192 FD_SET(data_ready_fd, &fdset);
193 int s = select(data_ready_fd + 1, &fdset, nullptr, nullptr, &tv);
194 if (s == 0) {
195 return 0;
196 } else if (s < 0) {
197 ESP_LOGE("dce", "select error %d", errno);
198 return -1;
199 }
200 if (FD_ISSET(data_ready_fd, &fdset)) {
201 ESP_LOGD("dce", "select read: modem data available");
202 return 1;
203 }
204 return -1;
205 }
