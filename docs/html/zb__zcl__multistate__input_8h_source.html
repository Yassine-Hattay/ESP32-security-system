338#define ZB_ZCL_MULTI_INPUT_SET_NORMAL_MODE(ep) \
339{ \
340 zb_uint8_t val; \
341 \
342 val = ZB_FALSE; \
343 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_MULTI_INPUT, \
344 ZB_ZCL_CLUSTER_SERVER_ROLE, \
345 ZB_ZCL_ATTR_MULTI_INPUT_OUT_OF_SERVICE_ID, \
346 &val, ZB_FALSE); \
347 \
348 val = ZB_ZCL_MULTI_INPUT_STATUS_FLAG_NORMAL; \
349 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_MULTI_INPUT, \
350 ZB_ZCL_CLUSTER_SERVER_ROLE, \
351 ZB_ZCL_ATTR_MULTI_INPUT_STATUS_FLAGS_ID, \

358#define ZB_ZCL_MULTI_INPUT_SET_OUT_OF_SERVICE(ep) \
359{ \
360 zb_zcl_attr_t *attr_desc; \
361 zb_uint8_t val; \
362 \
363 val = ZB_TRUE; \
364 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_MULTI_INPUT, \
365 ZB_ZCL_CLUSTER_SERVER_ROLE, \
366 ZB_ZCL_ATTR_MULTI_INPUT_OUT_OF_SERVICE_ID, \
367 &val, ZB_FALSE); \
368 \
369 attr_desc = zb_zcl_get_attr_desc_a(ep, ZB_ZCL_CLUSTER_ID_MULTI_INPUT, \
370 ZB_ZCL_CLUSTER_SERVER_ROLE, \
371 ZB_ZCL_ATTR_MULTI_INPUT_STATUS_FLAGS_ID); \
372 if (attr_desc) \
373 { \
374 val = *(zb_uint8_t*)attr_desc->data_p | ZB_ZCL_MULTI_INPUT_STATUS_FLAG_OUT_OF_SERVICE; \
375 ZB_ZCL_SET_ATTRIBUTE(ep, ZB_ZCL_CLUSTER_ID_MULTI_INPUT, \
376 ZB_ZCL_CLUSTER_SERVER_ROLE, \
377 ZB_ZCL_ATTR_MULTI_INPUT_STATUS_FLAGS_ID, \
378 &val, ZB_FALSE); \
379 } \
