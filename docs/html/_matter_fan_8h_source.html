15#pragma once
16#include <sdkconfig.h>
17#ifdef CONFIG_ESP_MATTER_ENABLE_DATA_MODEL
18
19#include <Matter.h>
20#include <MatterEndPoint.h>
21#include <app-common/zap-generated/cluster-objects.h>
22
23using namespace chip::app::Clusters::FanControl;
24
25// Matter Fan endpoint with On/Off, Mode and Speed control
26
27class MatterFan : public MatterEndPoint {
28public:
29 // Fan feature constants
30 static const uint8_t MAX_SPEED = 100; // maximum High speed
31 static const uint8_t MIN_SPEED = 1; // minimum Low speed
32 static const uint8_t OFF_SPEED = 0; // speed set by Matter when FAN_MODE_OFF
33
34 // Default Fan Modes: ON, SMART, HIGH and OFF
35
36 // Other mode will depend on what is the configured Fan Mode Sequence
37 enum FanMode_t {
38 FAN_MODE_OFF = (uint8_t)FanModeEnum::kOff,
39 FAN_MODE_LOW = (uint8_t)FanModeEnum::kLow,
40 FAN_MODE_MEDIUM = (uint8_t)FanModeEnum::kMedium,
41 FAN_MODE_HIGH = (uint8_t)FanModeEnum::kHigh,
42 FAN_MODE_ON = (uint8_t)FanModeEnum::kOn,
43 FAN_MODE_AUTO = (uint8_t)FanModeEnum::kAuto,
44 FAN_MODE_SMART = (uint8_t)FanModeEnum::kSmart
45 };
46
47 // Menu will always have ON, OFF, HIGH and SMART.
48 // AUTO will show up only when a AUTO SEQ is CONFIGURED
49 // LOW and MEDIUM depend on the SEQ MODE configuration
50 enum FanModeSequence_t {
51 FAN_MODE_SEQ_OFF_LOW_MED_HIGH = (uint8_t)FanModeSequenceEnum::kOffLowMedHigh,
52 FAN_MODE_SEQ_OFF_LOW_HIGH = (uint8_t)FanModeSequenceEnum::kOffLowHigh,
53 FAN_MODE_SEQ_OFF_LOW_MED_HIGH_AUTO = (uint8_t)FanModeSequenceEnum::kOffLowMedHighAuto,
54 FAN_MODE_SEQ_OFF_LOW_HIGH_AUTO = (uint8_t)FanModeSequenceEnum::kOffLowHighAuto,
55 FAN_MODE_SEQ_OFF_HIGH_AUTO = (uint8_t)FanModeSequenceEnum::kOffHighAuto,
56 FAN_MODE_SEQ_OFF_HIGH = (uint8_t)FanModeSequenceEnum::kOffHigh
57 };
58
59 MatterFan();
60 ~MatterFan();
61 virtual bool begin(uint8_t percent = 0, FanMode_t fanMode = FAN_MODE_OFF, FanModeSequence_t fanModeSeq = FAN_MODE_SEQ_OFF_HIGH);
62 void end(); // this will just stop processing Matter events
63
64 // returns a friendly string for the Fan Mode
65 static const char *getFanModeString(uint8_t mode) {
66 return fanModeString[mode];
67 }
68
69 // Fan Control of current On/Off state
70
71 bool setOnOff(bool newState, bool performUpdate = true); // sets Fan On/Off state
72 bool getOnOff(); // returns current Fan state
73 bool toggle(bool performUpdate = true); // toggle Fun On/Off state
74
75 // Fan Control of current speed percent
76
77 bool setSpeedPercent(uint8_t newPercent, bool performUpdate = true); // returns true if successful
78 uint8_t getSpeedPercent() { // returns current Fan Speed Percent
79 return currentPercent;
80 }
81
82 // Fan Control of current Fan Mode
83
84 bool setMode(FanMode_t newMode, bool performUpdate = true); // returns true if successful
85 FanMode_t getMode() { // returns current Fan Mode
86 return currentFanMode;
87 }
88 // used to update the state of the Fan using the current Matter Fan internal state
89 // It is necessary to set a user callback function using onChange() to handle the physical Fan motor state
90
91 void updateAccessory() {
92 if (_onChangeCB != NULL) {
93 _onChangeCB(currentFanMode, currentPercent);
94 }
95 }
96
97 // returns current Fan speed percent
98 operator uint8_t() {
99 return getSpeedPercent();
100 }
101
102 // User Callback for whenever the Fan Mode (state) is changed by the Matter Controller
103 using EndPointModeCB = std::function<bool(FanMode_t)>;
104 void onChangeMode(EndPointModeCB onChangeCB) {
105 _onChangeModeCB = onChangeCB;
106 }
107
108 // User Callback for whenever the Fan Speed Percentage value [0..100] is changed by the Matter Controller
109 using EndPointSpeedCB = std::function<bool(uint8_t)>;
110 void onChangeSpeedPercent(EndPointSpeedCB onChangeCB) {
111 _onChangeSpeedCB = onChangeCB;
112 }
113
114 // User Callback for whenever any parameter is changed by the Matter Controller
115 using EndPointCB = std::function<bool(FanMode_t, uint8_t)>;
116 void onChange(EndPointCB onChangeCB) {
117 _onChangeCB = onChangeCB;
118 }
119
120 // sets Fan speed percent
121 void operator=(uint8_t speedPercent) {
122 setSpeedPercent(speedPercent);
123 }
124
125 // this function is called by Matter internal event processor. It could be overwritten by the application, if necessary.
126 bool attributeChangeCB(uint16_t endpoint_id, uint32_t cluster_id, uint32_t attribute_id, esp_matter_attr_val_t *val);
127
128protected:
129 bool started = false;
130 uint8_t validFanModes = 0; // bitmap for valid Fan Modes - index of fanModeSequence[]
131
132 uint8_t currentPercent = 0; // current speed percent
133 FanMode_t currentFanMode = FAN_MODE_OFF; // current Fan Mode
134 EndPointModeCB _onChangeModeCB = NULL;
135 EndPointSpeedCB _onChangeSpeedCB = NULL;
136 EndPointCB _onChangeCB = NULL;
137
138 // bitmap for Fan Sequence Modes (OFF, LOW, MEDIUM, HIGH, AUTO)
139 static const uint8_t fanSeqModeOff = 0x01;
140 static const uint8_t fanSeqModeLow = 0x02;
141 static const uint8_t fanSeqModeMedium = 0x04;
142 static const uint8_t fanSeqModeHigh = 0x08;
143 static const uint8_t fanSeqModeOn = 0x10;
144 static const uint8_t fanSeqModeAuto = 0x20;
145 static const uint8_t fanSeqModeSmart = 0x40;
146
147 // bitmap for common modes: ON, OFF, HIGH and SMART
148 static const uint8_t fanSeqCommonModes = fanSeqModeOff | fanSeqModeOn | fanSeqModeHigh | fanSeqModeSmart;
149
150 static const uint8_t fanSeqModeOffLowMedHigh = fanSeqCommonModes | fanSeqModeLow | fanSeqModeMedium;
151 static const uint8_t fanSeqModeOffLowHigh = fanSeqCommonModes | fanSeqModeLow;
152 static const uint8_t fanSeqModeOffLowMedHighAuto = fanSeqCommonModes | fanSeqModeLow | fanSeqModeMedium | fanSeqModeAuto;
153 static const uint8_t fanSeqModeOffLowHighAuto = fanSeqCommonModes | fanSeqModeLow | fanSeqModeAuto;
154 static const uint8_t fanSeqModeOffHighAuto = fanSeqCommonModes | fanSeqModeAuto;
155 static const uint8_t fanSeqModeOffHigh = fanSeqCommonModes;
156
157 // bitmap for valid Fan Modes based on order defined in Zap Generated Cluster Enums
158 static const uint8_t fanModeSequence[6];
159
160 // string helper for the FAN MODE
161 static const char *fanModeString[7];
162};
163
164#endif /* CONFIG_ESP_MATTER_ENABLE_DATA_MODEL */
