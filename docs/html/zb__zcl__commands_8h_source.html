649#define ZB_ZCL_GENERAL_GET_NEXT_READ_ATTR_RES(data_buf, read_attr_resp) \
650{ \
651 zb_uint16_t resp_size = 0xffff; \
652 (read_attr_resp) = zb_buf_len(data_buf) >= ZB_ZCL_READ_ATTR_RESP_SIZE ? \
653 (zb_zcl_read_attr_res_t*)zb_buf_begin(data_buf) : NULL; \
654 \
655 if (read_attr_resp != NULL) \
656 { \
657 (read_attr_resp)->status = zb_zcl_zcl8_statuses_conversion((read_attr_resp)->status); \
658 resp_size = ZB_ZCL_READ_ATTR_RESP_SIZE; \
659 ZB_ZCL_HTOLE16_INPLACE(&(read_attr_resp)->attr_id); \
660 if ((read_attr_resp)->status == ZB_ZCL_STATUS_SUCCESS) \
661 { \
662 resp_size += \
663 sizeof(zb_uint8_t) + \
664 zb_zcl_get_attribute_size((read_attr_resp)->attr_type, (read_attr_resp)->attr_value); \
665 if (resp_size <= zb_buf_len(data_buf)) \
666 { \
667 ZB_ZCL_FIX_ENDIAN((read_attr_resp)->attr_value, (read_attr_resp)->attr_type); \
668 } \
669 } \
670 \
671 if (resp_size <= zb_buf_len(data_buf)) \
672 { \
673 (void)zb_buf_cut_left((data_buf), resp_size); \
674 } \
675 else \
676 { \
677 (read_attr_resp) = NULL; \
678 } \
679 } \
680}
