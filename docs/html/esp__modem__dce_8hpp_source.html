117 command_result pause_netif(bool do_pause, bool force = false, int delay = 1000)
118 {
119 command_result result;
120 if (do_pause) {
121 netif.pause();
122 Task::Delay(delay); // Mandatory 1s pause before
123 dte->set_command_callbacks();
124 result = device->set_command_mode();
125 } else {
126 result = device->resume_data_mode();
127 if (result == command_result::OK || force) {
128 netif.resume();
129 }
130 }
131 return result;
132 }
