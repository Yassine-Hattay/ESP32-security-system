216 struct command_cb {
217#ifdef CONFIG_ESP_MODEM_URC_HANDLER
218 got_line_cb urc_handler {};
219#endif
220 static const size_t GOT_LINE = SignalGroup::bit0;
221 got_line_cb got_line;
222 Lock line_lock{};
223 char separator{};
224 command_result result{};
225 SignalGroup signal;
226 bool process_line(uint8_t *data, size_t consumed, size_t len);
227 bool wait_for_line(uint32_t time_ms)
228 {
229 return signal.wait_any(command_cb::GOT_LINE, time_ms);
230 }
231 void set(got_line_cb l, char s = '\n')
232 {
233 Scoped<Lock> lock(line_lock);
234 if (l) {
235 // if we set the line callback, we have to reset the signal and the result
236 signal.clear(GOT_LINE);
237 result = command_result::TIMEOUT;
238 } else {
239 // if we clear the line callback, we check consistency (since we've locked the line processing)
240 if (signal.is_any(command_cb::GOT_LINE) && result == command_result::TIMEOUT) {
241 ESP_MODEM_THROW_IF_ERROR(ESP_ERR_INVALID_STATE);
242 }
243 }
244 got_line = std::move(l);
245 separator = s;
246 }
247 void give_up()
248 {
249 result = command_result::FAIL;
250 signal.set(GOT_LINE);
251 }
252 } command_cb;
253};
