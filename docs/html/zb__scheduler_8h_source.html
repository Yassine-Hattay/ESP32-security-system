266#define ZB_SCHED_WAIT_COND(condition) \
267do \
268{ \
269 ZB_SCHED_GLOBAL_LOCK(); \
270 while ( !(condition) ) \
271 { \
272 ZB_SCHED_GLOBAL_UNLOCK(); \
273 ZB_GO_IDLE(); \
274 ZB_SCHED_GLOBAL_LOCK(); \
275 } \
276 ZB_SCHED_GLOBAL_UNLOCK(); \
277} \
278while(0)
