31/*
32 * This file contains compile-time flags that can override the
33 * autodetection performed in relevant files. Each flag is a macro; it
34 * deactivates the feature if defined to 0, activates it if defined to a
35 * non-zero integer (normally 1). If the macro is not defined, then
36 * autodetection applies.
37 */
38
39/*
40 * When BR_64 is enabled, 64-bit integer types are assumed to be
41 * efficient (i.e. the architecture has 64-bit registers and can
42 * do 64-bit operations as fast as 32-bit operations).
43 *
44#define BR_64 1
45 */
46
47/*
48 * When BR_LOMUL is enabled, then multiplications of 32-bit values whose
49 * result are truncated to the low 32 bits are assumed to be
50 * substantially more efficient than 32-bit multiplications that yield
51 * 64-bit results. This is typically the case on low-end ARM Cortex M
52 * systems (M0, M0+, M1, and arguably M3 and M4 as well).
53#define BR_LOMUL 1
54 */
55
56/*
57 * When BR_SLOW_MUL is enabled, multiplications are assumed to be
58 * substantially slow with regards to other integer operations, thus
59 * making it worth to make more operations for a given task if it allows
60 * using less multiplications.
61 *
62#define BR_SLOW_MUL 1
63 */
64
65/*
66 * When BR_SLOW_MUL15 is enabled, short multplications (on 15-bit words)
67 * are assumed to be substantially slow with regards to other integer
68 * operations, thus making it worth to make more integer operations if
69 * it allows using less multiplications.
70 *
71#define BR_SLOW_MUL15 1
72 */
73
74/*
75 * When BR_CT_MUL31 is enabled, multiplications of 31-bit values (used
76 * in the "i31" big integer implementation) use an alternate implementation
77 * which is slower and larger than the normal multiplication, but should
78 * ensure constant-time multiplications even on architectures where the
79 * multiplication opcode takes a variable number of cycles to complete.
80 *
81#define BR_CT_MUL31 1
82 */
83
84/*
85 * When BR_CT_MUL15 is enabled, multiplications of 15-bit values (held
86 * in 32-bit words) use an alternate implementation which is slower and
87 * larger than the normal multiplication, but should ensure
88 * constant-time multiplications on most/all architectures where the
89 * basic multiplication is not constant-time.
90#define BR_CT_MUL15 1
91 */
92
93/*
94 * When BR_NO_ARITH_SHIFT is enabled, arithmetic right shifts (with sign
95 * extension) are performed with a sequence of operations which is bigger
96 * and slower than a simple right shift on a signed value. This avoids
97 * relying on an implementation-defined behaviour. However, most if not
98 * all C compilers use sign extension for right shifts on signed values,
99 * so this alternate macro is disabled by default.
100#define BR_NO_ARITH_SHIFT 1
101 */
102
103/*
104 * When BR_RDRAND is enabled, the SSL engine will use the RDRAND opcode
105 * to automatically obtain quality randomness for seeding its internal
106 * PRNG. Since that opcode is present only in recent x86 CPU, its
107 * support is dynamically tested; if the current CPU does not support
108 * it, then another random source will be used, such as /dev/urandom or
109 * CryptGenRandom().
110 *
111#define BR_RDRAND 1
112 */
113
114/*
115 * When BR_USE_GETENTROPY is enabled, the SSL engine will use the
116 * getentropy() function to obtain quality randomness for seeding its
117 * internal PRNG. On Linux and FreeBSD, getentropy() is implemented by
118 * the standard library with the system call getrandom(); on OpenBSD,
119 * getentropy() is the system call, and there is no getrandom() wrapper,
120 * hence the use of the getentropy() function for maximum portability.
121 *
122 * If the getentropy() call fails, and BR_USE_URANDOM is not explicitly
123 * disabled, then /dev/urandom will be used as a fallback mechanism. On
124 * FreeBSD and OpenBSD, this does not change much, since /dev/urandom
125 * will block if not enough entropy has been obtained since last boot.
126 * On Linux, /dev/urandom might not block, which can be troublesome in
127 * early boot stages, which is why getentropy() is preferred.
128 *
129#define BR_USE_GETENTROPY 1
130 */
131
132/*
133 * When BR_USE_URANDOM is enabled, the SSL engine will use /dev/urandom
134 * to automatically obtain quality randomness for seeding its internal
135 * PRNG.
136 *
137#define BR_USE_URANDOM 1
138 */
139
140/*
141 * When BR_USE_WIN32_RAND is enabled, the SSL engine will use the Win32
142 * (CryptoAPI) functions (CryptAcquireContext(), CryptGenRandom()...) to
143 * automatically obtain quality randomness for seeding its internal PRNG.
144 *
145 * Note: if both BR_USE_URANDOM and BR_USE_WIN32_RAND are defined, the
146 * former takes precedence.
147 *
148#define BR_USE_WIN32_RAND 1
149 */
150
151/*
152 * When BR_USE_UNIX_TIME is enabled, the X.509 validation engine obtains
153 * the current time from the OS by calling time(), and assuming that the
154 * returned value (a 'time_t') is an integer that counts time in seconds
155 * since the Unix Epoch (Jan 1st, 1970, 00:00 UTC).
156 */
157#define BR_USE_UNIX_TIME 0
158
159
160/*
161 * When BR_USE_WIN32_TIME is enabled, the X.509 validation engine obtains
162 * the current time from the OS by calling the Win32 function
163 * GetSystemTimeAsFileTime().
164 *
165 * Note: if both BR_USE_UNIX_TIME and BR_USE_WIN32_TIME are defined, the
166 * former takes precedence.
167 *
168#define BR_USE_WIN32_TIME 1
169 */
170
171/*
172 * When BR_ARMEL_CORTEXM_GCC is enabled, some operations are replaced with
173 * inline assembly which is shorter and/or faster. This should be used
174 * only when all of the following are true:
175 * - target architecture is ARM in Thumb mode
176 * - target endianness is little-endian
177 * - compiler is GCC (or GCC-compatible for inline assembly syntax)
178 *
179 * This is meant for the low-end cores (Cortex M0, M0+, M1, M3).
180 * Note: if BR_LOMUL is not explicitly enabled or disabled, then
181 * enabling BR_ARMEL_CORTEXM_GCC also enables BR_LOMUL.
182 */
183#if defined(__arm__) && defined(__thumb__)
184#define BR_ARMEL_CORTEXM_GCC 1
185#endif
186
187/*
188 * When BR_AES_X86NI is enabled, the AES implementation using the x86 "NI"
189 * instructions (dedicated AES opcodes) will be compiled. If this is not
190 * enabled explicitly, then that AES implementation will be compiled only
191 * if a compatible compiler is detected. If set explicitly to 0, the
192 * implementation will not be compiled at all.
193 *
194#define BR_AES_X86NI 1
195 */
196
197/*
198 * When BR_SSE2 is enabled, SSE2 intrinsics will be used for some
199 * algorithm implementations that use them (e.g. chacha20_sse2). If this
200 * is not enabled explicitly, then support for SSE2 intrinsics will be
201 * automatically detected. If set explicitly to 0, then SSE2 code will
202 * not be compiled at all.
203 *
204#define BR_SSE2 1
205 */
206
207/*
208 * When BR_POWER8 is enabled, the AES implementation using the POWER ISA
209 * 2.07 opcodes (available on POWER8 processors and later) is compiled.
210 * If this is not enabled explicitly, then that implementation will be
211 * compiled only if a compatible compiler is detected, _and_ the target
212 * architecture is POWER8 or later.
213 *
214#define BR_POWER8 1
215 */
216
217/*
218 * When BR_INT128 is enabled, then code using the 'unsigned __int64'
219 * and 'unsigned __int128' types will be used to leverage 64x64->128
220 * unsigned multiplications. This should work with GCC and compatible
221 * compilers on 64-bit architectures.
222 *
223#define BR_INT128 1
224 */
225
226/*
227 * When BR_UMUL128 is enabled, then code using the '_umul128()' and
228 * '_addcarry_u64()' intrinsics will be used to implement 64x64->128
229 * unsigned multiplications. This should work on Visual C on x64 systems.
230 *
231#define BR_UMUL128 1
232 */
233
234/*
235 * When BR_LE_UNALIGNED is enabled, then the current architecture is
236 * assumed to use little-endian encoding for integers, and to tolerate
237 * unaligned accesses with no or minimal time penalty.
238 *
239#define BR_LE_UNALIGNED 1
240 */
241
242/*
243 * When BR_BE_UNALIGNED is enabled, then the current architecture is
244 * assumed to use big-endian encoding for integers, and to tolerate
245 * unaligned accesses with no or minimal time penalty.
246 *
247#define BR_BE_UNALIGNED 1
248 */
