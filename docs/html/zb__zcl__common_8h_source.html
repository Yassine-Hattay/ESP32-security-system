554static ZB_INLINE zb_uint16_t zb_zcl_string_append_data(zb_uint8_t *zcl_str,
555 zb_uint8_t zcl_str_max_size,
556 const zb_uint8_t *c_str,
557 zb_uint8_t c_str_len)
558{
559 zb_uint16_t newlen = ZB_ZCL_GET_STRING_LENGTH(zcl_str) + (zb_uint16_t)c_str_len;
560
561 if (newlen < (zb_uint16_t) zcl_str_max_size)
562 {
563 ZB_MEMCPY(ZB_ZCL_GET_STRING_END(zcl_str), c_str, c_str_len);
564 ZB_ZCL_SET_STRING_LENGTH(zcl_str, (zb_uint8_t)newlen);
565 }
566 else
567 {
568 newlen = 0;
569 }
570
571 return newlen;
572}

575static ZB_INLINE zb_uint16_t zb_zcl_string_append_byte(zb_uint8_t *zcl_str,
576 zb_uint8_t zcl_str_max_size,
577 zb_uint8_t value)
578{
579 zb_uint16_t newlen = (zb_uint16_t)ZB_ZCL_GET_STRING_LENGTH(zcl_str) + 1U;
580
581 if (newlen < (zb_uint16_t) zcl_str_max_size)
582 {
583 *ZB_ZCL_GET_STRING_END(zcl_str) = value;
584 ZB_ASSERT(newlen <= ZB_UINT8_MAX);
585 ZB_ZCL_SET_STRING_LENGTH(zcl_str, (zb_uint8_t)newlen);
586 }
587 else
588 {
589 newlen = 0;
590 }
591
592 return newlen;
593}
