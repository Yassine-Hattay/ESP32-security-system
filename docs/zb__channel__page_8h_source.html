207/* Numbers [23 - 31] have common first bit
208 *
209 * 23 = 0b10111
210 * 24 = 0b11000
211 * 25 = 0b11001
212 * 26 = 0b11010
213 * 27 = 0b11011
214 * 28 = 0b11100
215 * 29 = 0b11101
216 * 30 = 0b11110
217 * 31 = 0b11111
218 *
219 * So, it is enough to compare it with 0b10111 mask = 0x17 = 23 (dec).
220 */
221#define ZB_LOGICAL_PAGE_SUB_GHZ_PAGE_MASK ZB_CHANNEL_PAGE23_SUB_GHZ
222
223#define ZB_LOGICAL_PAGE_IS_SUB_GHZ(logical_page) \
224 (ZB_U2B((logical_page) & ZB_LOGICAL_PAGE_SUB_GHZ_PAGE_MASK))
225
226#define ZB_LOGICAL_PAGE_IS_SUB_GHZ_NA_FSK(logical_page) \
227 ((logical_page) == ZB_CHANNEL_PAGE23_SUB_GHZ)
228
229#define ZB_LOGICAL_PAGE_IS_SUB_GHZ_EU_FSK(logical_page) \
230 ((logical_page) >= ZB_CHANNEL_PAGE24_SUB_GHZ && (logical_page) <= ZB_CHANNEL_PAGE27_SUB_GHZ)
231
232#define ZB_LOGICAL_PAGE_IS_SUB_GHZ_GB_FSK(logical_page) \
233 ((logical_page) >= ZB_CHANNEL_PAGE28_SUB_GHZ && (logical_page) <= ZB_CHANNEL_PAGE31_SUB_GHZ)
234
235#define ZB_LOGICAL_PAGE_IS_2_4GHZ(logical_page) \
236 ((logical_page) == ZB_CHANNEL_PAGE0_2_4_GHZ)
237
238#define ZB_CHANNEL_PAGE_IS_SUB_GHZ(channel_page) \
239 (ZB_LOGICAL_PAGE_IS_SUB_GHZ(ZB_CHANNEL_PAGE_GET_PAGE(channel_page)))
