67 {
68 if (mode == modem_mode::DATA_MODE) {
69 if (set_data_mode() != command_result::OK) {
70 return resume_data_mode() == command_result::OK;
71 }
72 return true;
73 } else if (mode == modem_mode::COMMAND_MODE) {
74 Task::Delay(1000); // Mandatory 1s pause before
75 int retry = 0;
76 while (retry++ < 3) {
77 if (set_command_mode() == command_result::OK) {
78 return true;
79 }
80 // send a newline to delimit the escape from the upcoming sync command
81 uint8_t delim = '\n';
82 dte->write(&delim, 1);
83 if (sync() == command_result::OK) {
84 return true;
85 }
86 Task::Delay(1000); // Mandatory 1s pause before escape
87 }
88 return false;
89 } else if (mode == modem_mode::CMUX_MODE) {
90 return set_cmux() == command_result::OK;
91 }
92 return true;
