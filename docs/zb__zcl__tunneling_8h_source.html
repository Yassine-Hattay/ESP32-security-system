421#define ZB_ZCL_TUNNELING_SEND_REQUEST_TUNNEL( \
422 _param, _addr, _dst_addr_mode, _dst_ep, _ep, _prfl_id, _def_resp, _cb, \
423 _protocol_id, _manufacturer_code, _flow_control_support, \
424 _max_incoming_transfer_size) \
425{ \
426 zb_zcl_tunneling_set_max_incoming_to_cli_transfer_size( \
427 _max_incoming_transfer_size); \
428 { \
429 zb_bufid_t _buffer = _param; \
430 zb_uint8_t* __ptr = zb_zcl_start_command_header(_buffer, \
431 ZB_ZCL_CONSTRUCT_FRAME_CONTROL(ZB_ZCL_FRAME_TYPE_CLUSTER_SPECIFIC, \
432 ZB_ZCL_NOT_MANUFACTURER_SPECIFIC, \
433 ZB_ZCL_FRAME_DIRECTION_TO_SRV, (_def_resp)), \
434 0, /* No manuf_code */ \
435 ZB_ZCL_TUNNELING_CLI_CMD_REQUEST_TUNNEL, NULL); \
436 ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_protocol_id)); \
437 ZB_ZCL_PACKET_PUT_DATA16_VAL(__ptr, (_manufacturer_code)); \
438 ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_flow_control_support)); \
439 ZB_ZCL_PACKET_PUT_DATA16_VAL(__ptr, (_max_incoming_transfer_size)); \
440 ZB_ZCL_FINISH_PACKET(_buffer, __ptr) \
441 ZB_ZCL_SEND_COMMAND_SHORT( \
442 _buffer, (_addr), (_dst_addr_mode), (_dst_ep), (_ep), (_prfl_id), \
443 ZB_ZCL_CLUSTER_ID_TUNNELING, (_cb)); \
444 } \
445}
